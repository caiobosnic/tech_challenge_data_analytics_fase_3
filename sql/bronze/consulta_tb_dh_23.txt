--Análise da base tb_dh23--
--1. Quantidade de registros--
SELECT COUNT (*) AS qtd_registros_23
from tb_dh_23;
-- 5293 Respostas da pesquisa--

--2. Quantidade de Colunas (não perguntas)--
select count(*) as qtd_colunas_23
from information_schema.columns
where table_name = 'tb_dh_23'
-- 399 colunas (nao perguntas)--

--3. registros sem id--
SELECT
    COUNT(*) AS total_registros,
    SUM(
        CASE
            WHEN "_'p0'__'id'_#0" IS NULL
              OR TRIM(CAST("_'p0'__'id'_#0" AS VARCHAR)) = ''
            THEN 1
            ELSE 0
        END
    ) AS ids_nulos
FROM tb_dh_23;
-- Não há nenhuma linha com ID nulo/vazio--

--4. verificação de ID duplicado--
SELECT
    "_'p0'__'id'_#0" AS id,
    COUNT(*) AS quantidade
FROM tb_dh_23
WHERE "_'p0'__'id'_#0" IS NOT NULL
GROUP BY "_'p0'__'id'_#0"
HAVING COUNT(*) > 1
ORDER BY quantidade DESC;
-- não há IDs duplicados--

--5.Verificação se a quantidade de IDs únicas é a mesma de registros--
SELECT
    COUNT(*) AS total_registros,
    COUNT(DISTINCT "_'p0'__'id'_#0") AS ids_unicos
FROM tb_dh_23;
show columns from tb_dh_23
-- ids_unicos = qtd_registros -> entende-se que não correspondentes duplicados--

--6. Verificação dos tipos de variáveis--
SELECT
    data_type,
    COUNT(*) AS quantidade_colunas
FROM information_schema.columns
WHERE table_schema = 'default'
  AND table_name = 'tb_dh_23'
GROUP BY data_type
ORDER BY quantidade_colunas DESC;
-- Todas variáveis varchar--

--15. Uso básico de IA--
SELECT
    CASE
        WHEN "_'p4_m_'__'utiliza_chatgpt_ou_llms_no_trabalho?'_#238"
             LIKE '%Não utilizo nenhum tipo%'
            THEN 'Não utiliza IA Generativa'
        ELSE 'Utiliza IA Generativa'
    END AS uso_ia,
    COUNT(*) AS quantidade,
    ROUND(
        100.0 * COUNT(*) / SUM(COUNT(*)) OVER (),
        2
    ) AS percentual
FROM tb_dh_23
WHERE "_'p4_m_'__'utiliza_chatgpt_ou_llms_no_trabalho?'_#238" IS NOT NULL
GROUP BY
    CASE
        WHEN "_'p4_m_'__'utiliza_chatgpt_ou_llms_no_trabalho?'_#238"
             LIKE '%Não utilizo nenhum tipo%'
            THEN 'Não utiliza IA Generativa'
        ELSE 'Utiliza IA Generativa'
    END
ORDER BY quantidade DESC;
--baseado nessa classificação de uso ou nao de IA, temos que 86% dos profissionais usam IAs em alguma parte do trabalho, enquanto os outros 14% não fazer uso de IA--
